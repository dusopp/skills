# agent-guardrails deny list
# =========================================================================
# One regex per line. Blank lines and lines starting with # are ignored.
# Section headers route patterns:
#   [shell]        matched against shell command strings (Claude Bash/PowerShell
#                  tools, Copilot terminal tools) AND against any string found
#                  under a property named "command" inside MCP tool arguments.
#   [mcp-tool]     matched against MCP tool NAMES only (mcp__server__tool).
#   [mcp-command]  matched against MCP-carried "command" field values only.
#
# Regex rules (this is the .NET flavor, evaluated case-INsensitively):
#   - NO POSIX classes: [[:alpha:]] fails SILENTLY in .NET. Use \w, \s, \b.
#   - Use \b for word boundaries, never \< or \>.
#   - ^ and $ match the whole (possibly multi-line) command string, not lines.
#   - (?!\S) means "end of token": accepts whitespace or end-of-string, and
#     rejects longer flags (--force matches, --force-with-lease does not).
#   - [^\n;&|]* means "within the same shell statement": prevents a pattern
#     from matching across ; && || | boundaries. Prefer it over .*
#   - A trailing space on a line is a literal regex character: lines are
#     trimmed on load, so do not rely on trailing whitespace.
#
# Design rule: block only IRREVERSIBLE or CATASTROPHIC actions (mass data
# loss, purges that defeat soft-delete, safety-net removal, identity
# destruction, irreversible publishes). Local-destructive-but-recoverable
# commands stay ALLOWED - over-blocking kills agent usefulness.
#
# Deliberately ALLOWED (do not "fix" these): az webapp/functionapp delete,
# az sql db delete, az keyvault delete (soft delete is recoverable; purge is
# not), single az storage container/blob delete, dotnet ef database drop,
# git push --force-with-lease, rm -rf node_modules, git clean, and all
# Azure DevOps MCP *_write tools (daily workflow; that server ships no
# delete tools by design as of v2.9).
#
# Known accepted gaps (documented, not bugs): git -C <dir> push --force
# (options between git and push), combined short flags like -fu, split rm
# flags (rm -r -f /), quoting/encoding obfuscation. This guard is a seatbelt
# against accidents, not a sandbox against a malicious agent.
# =========================================================================

[mcp-tool]
# Matched against MCP tool names only (mcp__server__tool, mcp__plugin_x_y__tool).
# Verb suffix match, not exact names: server prefixes vary per machine/project.
__[\w.-]*(delete|purge|destroy)
# Old Azure MCP servers shipped arbitrary az/azd execution tools; block them
# defensively (harmless on current servers, critical on pinned old versions).
\bextension[-_]azd?\b

[mcp-command]
# Matched only against "command" field values inside MCP tool arguments
# (e.g. Azure MCP namespace mode routes the real operation through
# tool_input.command). Never matched against free-text fields.
\b(delete|purge|destroy)\b
